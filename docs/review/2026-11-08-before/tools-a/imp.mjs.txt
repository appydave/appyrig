import fs from 'node:fs'; import path from 'node:path';
const [tf, prod] = process.argv.slice(2).map(f => fs.readFileSync(f,'utf8').trim().split('\n'));
const tests = [...tf, 'client/src/pages/page-test-kit.ts','client/src/test/fleetFixtures.ts','client/src/test/queueFixtures.ts'];
const direct = new Set();
const norm = p => p.replace(/\.(tsx?|js)$/,'').replace(/\/index$/,'/index');
for (const t of tests) { const src = fs.readFileSync(t,'utf8');
  for (const m of src.matchAll(/(?:from|import\()\s*['"]([^'"]+)['"]/g)) { let s=m[1]; let r;
    if (s.startsWith('.')) r = path.normalize(path.join(path.dirname(t), s)); else if (s.startsWith('@client/')) r='client/src/'+s.slice(8); else continue;
    r = norm(r); direct.add(r); direct.add(r+'/index'); } }
const out=[]; for (const p of prod){ const k=norm(p); if(!direct.has(k)) out.push(fs.readFileSync(p,'utf8').split('\n').length+' '+p); }
out.sort((a,b)=>parseInt(b)-parseInt(a)); console.log(out.length+' prod files not directly imported by any test'); console.log(out.join('\n'));
